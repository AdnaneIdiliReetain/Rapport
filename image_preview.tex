% For borders around images
\newcommand{\borderedimage}[3][0.9\textwidth]{%
  \begin{figure}
    \centering
    \setlength{\fboxsep}{5pt}%
    \setlength{\fboxrule}{1pt}%
    \fbox{\includegraphics[width=#1]{#2}}
    \caption{#3}
  \end{figure}
}

% For simple centered images
\newcommand{\simpleimage}[3][0.9\textwidth]{%
  \begin{figure}
    \centering
    \includegraphics[width=#1]{#2}
    \caption{#3}
  \end{figure}
}

% For colored border images
\newcommand{\colorimage}[4][0.9\textwidth]{%
  \begin{figure}
    \centering
    \setlength{\fboxsep}{5pt}%
    \setlength{\fboxrule}{1pt}%
    \fcolorbox{#4}{white}{\includegraphics[width=#1]{#2}}
    \caption{#3}
  \end{figure}
}
